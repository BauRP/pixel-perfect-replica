cmake_minimum_required(VERSION 3.22)
project(soundworld_sonivox LANGUAGES C CXX)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "Android ARM64 required")
endif()
if(NOT EXISTS "${SONIVOX_SOURCE}/LICENSE" OR NOT EXISTS "${SONIVOX_SOURCE}/NOTICE")
  message(FATAL_ERROR "Reviewed Sonivox checkout required")
endif()
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_SONIVOX_STATIC ON CACHE BOOL "" FORCE)
set(BUILD_SONIVOX_SHARED OFF CACHE BOOL "" FORCE)
add_subdirectory("${SONIVOX_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/sonivox" EXCLUDE_FROM_ALL)
add_library(soundworld_sonivox SHARED sonivox_pcm.cpp)
target_compile_features(soundworld_sonivox PRIVATE cxx_std_17)
target_include_directories(soundworld_sonivox PRIVATE "${SONIVOX_SOURCE}/arm-wt-22k/host_src")
target_link_libraries(soundworld_sonivox PRIVATE sonivox-static)
target_link_options(soundworld_sonivox PRIVATE "-Wl,-z,max-page-size=16384")
