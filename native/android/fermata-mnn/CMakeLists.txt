cmake_minimum_required(VERSION 3.22)
project(soundworld_fermata_mnn LANGUAGES C CXX)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "Build the actual Android ARM64 bridge")
endif()
if(NOT EXISTS "${MNN_SOURCE}/transformers/llm/engine/include/llm/llm.hpp")
  message(FATAL_ERROR "MNN 3.6.1 source checkout required")
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(MNN_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(MNN_SEP_BUILD OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_LLM ON CACHE BOOL "" FORCE)
set(MNN_BUILD_LLM_OMNI OFF CACHE BOOL "" FORCE)
set(MNN_SUPPORT_TRANSFORMER_FUSE ON CACHE BOOL "" FORCE)
set(MNN_LOW_MEMORY ON CACHE BOOL "" FORCE)
set(MNN_CPU_WEIGHT_DEQUANT_GEMM ON CACHE BOOL "" FORCE)
set(MNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_BENCHMARK OFF CACHE BOOL "" FORCE)
set(MNN_LLM_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(LLM_SUPPORT_HTTP_RESOURCE OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_FOR_ANDROID_COMMAND ON CACHE BOOL "" FORCE)
set(MNN_OPENCL OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_OPENCV OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_AUDIO OFF CACHE BOOL "" FORCE)
add_subdirectory("${MNN_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/mnn" EXCLUDE_FROM_ALL)
add_library(soundworld_fermata_mnn SHARED fermata_mnn.cpp)
target_compile_features(soundworld_fermata_mnn PRIVATE cxx_std_17)
target_include_directories(soundworld_fermata_mnn PRIVATE
  "${MNN_SOURCE}/include" "${MNN_SOURCE}/transformers/llm/engine/include")
target_link_libraries(soundworld_fermata_mnn PRIVATE MNN log android)
target_link_options(soundworld_fermata_mnn PRIVATE "-Wl,-z,max-page-size=16384")
